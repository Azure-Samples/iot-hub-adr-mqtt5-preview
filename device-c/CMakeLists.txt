cmake_minimum_required(VERSION 3.21)
project(mqttv5_device_c LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(MQTTV5_SDK_ROOT "" CACHE PATH "Pinned azure-iot-sdk checkout")
set(MQTTV5_DEPENDENCIES "" CACHE PATH "Existing FetchContent source cache")
set(MQTTV5_SDK_COMMIT "62f7f9580c67a8275f3c92287919ba40c01d472c")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MQTTV5_SDK_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE sdk_head OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT sdk_head STREQUAL MQTTV5_SDK_COMMIT)
    message(FATAL_ERROR "SDK must be at tested commit ${MQTTV5_SDK_COMMIT}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MQTTV5_SDK_ROOT}" diff --quiet HEAD -- c
    RESULT_VARIABLE sdk_dirty)
if(NOT sdk_dirty EQUAL 0)
    message(FATAL_ERROR "The pinned SDK c tree must have no tracked modifications")
endif()
foreach(dep azure_sdk_for_c paho_mqtt_c)
    if(NOT EXISTS "${MQTTV5_DEPENDENCIES}/${dep}-src/CMakeLists.txt")
        message(FATAL_ERROR "Missing existing dependency source: ${MQTTV5_DEPENDENCIES}/${dep}-src")
    endif()
    string(TOUPPER "${dep}" dep_upper)
    set(FETCHCONTENT_SOURCE_DIR_${dep_upper} "${MQTTV5_DEPENDENCIES}/${dep}-src" CACHE PATH "" FORCE)
endforeach()
set(ENV{AZURE_SDK_DISABLE_AUTO_VCPKG} 1)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_MSVC_STATIC_RT TRUE)
find_package(OpenSSL 3.0 REQUIRED)
set(AZ_IOT_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(AZ_IOT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AZ_IOT_BUILD_E2E OFF CACHE BOOL "" FORCE)
set(AZ_IOT_INSTALL OFF CACHE BOOL "" FORCE)
set(AZ_IOT_WITH_PAHO ON CACHE BOOL "" FORCE)
set(AZ_IOT_WITH_AZ_MQTT OFF CACHE BOOL "" FORCE)
set(AZ_IOT_WITH_CRYPTO_OPENSSL ON CACHE BOOL "" FORCE)
set(AZ_IOT_WITH_CRYPTO_MBEDTLS OFF CACHE BOOL "" FORCE)
set(AZ_IOT_WITH_CERT_PROVIDER_MANAGED ON CACHE BOOL "" FORCE)
add_subdirectory("${MQTTV5_SDK_ROOT}/c" sdk)
get_property(paho_patched GLOBAL PROPERTY AZ_IOT_PAHO_PATCHED)
if(NOT paho_patched)
    message(FATAL_ERROR "Prepare Paho v1.3.16 with the pinned SDK's patches before building. See device-c/README.md.")
endif()

add_executable(mqttv5-device main.c)
target_link_libraries(mqttv5-device PRIVATE az_iot_mqttv5 az_iot_adapter_paho
    az_iot_crypto_openssl az_iot_certificate_provider_managed)
if(WIN32)
    target_compile_definitions(mqttv5-device PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(mqttv5-device PRIVATE advapi32)
endif()
if(MSVC)
    target_compile_options(mqttv5-device PRIVATE /W4 /WX)
else()
    target_compile_options(mqttv5-device PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME device-c-offline COMMAND mqttv5-device --self-test)
